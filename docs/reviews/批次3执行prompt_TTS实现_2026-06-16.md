         """接收 DashScope 返回的 TTS 音频片段，按 response_id 关联到对应段并下发"""
         if not ev.audio or not ev.response_id:
             return
         seg = self._by_response.get(ev.response_id)
         if seg is None:
             return
         
         # 累积音频（若一个 response 分多次发 audio.delta）
         # 或直接发（若 DashScope 一次发完）
         await self.emit({
             "type": "audio_segment",
             "segmentId": seg.segment_id,

     ```typescript
     export function useTtsPlayback(options: { onError?: (msg: string) => void }) {
       const context = ref<AudioContext | null>(null);
       const gainNode = ref<GainNode | null>(null);
       const queue: Array<{ segmentId: string; buffer: AudioBuffer }> = [];
       let nextStartTime = 0;
       
       async function init(sampleRate: number) {
         if (context.value) return;
         context.value = new AudioContext({ sampleRate });
         gainNode.value = context.value.createGain();
         gainNode.value.connect(context.value.destination);
       }
       
       async function enqueue(segmentId: string, audioBase64: string, sampleRate: number) {
         await init(sampleRate);
         const arrayBuffer = base64ToArrayBuffer(audioBase64);
         const audioBuffer = await context.value!.decodeAudioData(arrayBuffer);
         
         const source = context.value!.createBufferSource();
         source.buffer = audioBuffer;
         source.connect(gainNode.value!);
         
         const now = context.value!.currentTime;
         const startAt = Math.max(now, nextStartTime);
         source.start(startAt);
         nextStartTime = startAt + audioBuffer.duration;
       }
       
       function setVolume(volume: number) {
         if (gainNode.value) gainNode.value.gain.value = volume;
       }
       
       function mute() { setVolume(0); }
       function unmute() { setVolume(1); }
       
       return { enqueue, setVolume, mute, unmute };
     }
     ```

## 批次3实际落地记录

### 实现摘要

- 前端 `CreateSessionPayload`、`QuickFormState`、`FloatingFormState` 增加 `ttsEnabled`，快速设置面板和高级设置弹窗均提供“语音播报”开关，默认关闭。
- 后端 `pipeline` 接收 `response.audio.delta`，下发统一 `audio_segment` 事件；真实链路发现 audio delta 可能早于文本绑定，因此增加 response_id 维度的 pending audio 缓存，待译文绑定后再 flush。
- 前端新增 `createTtsPlayback`，按后端 `audio_segment.sampleRate` 创建 `AudioContext`，将 PCM16 base64 转为 `AudioBuffer`，用 `GainNode` 顺序播放。
- `MediaPanel` 增加静音按钮和音量 slider；store 在结束、重置、进入报告时停止 TTS 队列。
- LiveTranslate 真实链路验证发现 `qwen3.5-livetranslate-flash-realtime` 不支持旧配置 `voice=Cherry`，已将 LiveTranslate 默认音色改为 `Tina`，并在 provider 中兼容旧 `.env` 的 `Cherry -> Tina`。

### 真实链路采样

- 输入：`frontend/public/fixtures/test-video/video.mp4`
- 配置：`MODEL_PROVIDER=real`，`tts_enabled=True`
- 结果：成功收到 `audio_segment` 事件 161 个；首个音频帧 `sampleRate=24000`、`bytes=15360`、按 24kHz PCM16 推算约 `320.0ms`。
- 结论：DashScope LiveTranslate TTS 输出为 PCM 24kHz，前端播放层按 24kHz 解码，避免采样率不匹配导致爆音。
